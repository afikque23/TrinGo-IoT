#include <Arduino.h>
#include <TinyGPSPlus.h>
#include <HardwareSerial.h>
#include <Wire.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_BMP280.h>

// Buat objek TinyGPSPlus
TinyGPSPlus gps;

// Buat objek BMP280 (I2C)
Adafruit_BMP280 bmp;

// ESP32 menggunakan HardwareSerial, kita gunakan Serial2
// RX Pin ESP32 = 16 (Hubungkan ke TX NEO-7M)
// TX Pin ESP32 = 17 (Hubungkan ke RX NEO-7M)
#define RXD2 16
#define TXD2 17

void setup() {
  // Serial Monitor ke Komputer
  Serial.begin(115200);
  Serial.println("Memulai Tes GPS NEO-7M...");
  
  // Serial2 untuk komunikasi dengan GPS Module
  // NEO-7M secara default menggunakan baud rate 9600
  Serial2.begin(9600, SERIAL_8N1, RXD2, TXD2);
  
  // Inisialisasi BMP280
  // Coba alamat 0x76, jika gagal coba 0x77
  if (!bmp.begin(0x76) && !bmp.begin(0x77)) {
    Serial.println("ERROR: Sensor BMP280 tidak terdeteksi! Cek koneksi I2C (SDA=21, SCL=22).");
  } else {
    // Konfigurasi default BMP280 agar stabil
    bmp.setSampling(Adafruit_BMP280::MODE_NORMAL,     
                    Adafruit_BMP280::SAMPLING_X2,     
                    Adafruit_BMP280::SAMPLING_X16,    
                    Adafruit_BMP280::FILTER_X16,      
                    Adafruit_BMP280::STANDBY_MS_500); 
    Serial.println("Sensor BMP280 Berhasil Diinisialisasi!");
  }
  
  Serial.println("Menunggu sinyal GPS (Pastikan antena GPS menghadap langit)...");
}

void loop() {
  // Selama ada data yang masuk dari modul GPS
  while (Serial2.available() > 0) {
    char c = Serial2.read();
    gps.encode(c);
  }

  // Jika data lokasi sudah diperbarui (mendapat sinyal fix)
  if (gps.location.isUpdated()) {
    Serial.print("Sinyal DIDAPAT! | ");
    Serial.print("Satelit: ");
    Serial.print(gps.satellites.value());
    Serial.print(" | Lat: ");
    Serial.print(gps.location.lat(), 6);
    Serial.print(" | Lng: ");
    Serial.print(gps.location.lng(), 6);
    
    // Membaca data dari BMP280
    Serial.print(" | Suhu: ");
    Serial.print(bmp.readTemperature());
    Serial.print(" *C");
    Serial.print(" | Tekanan: ");
    Serial.print(bmp.readPressure() / 100.0F); // hPa
    Serial.print(" hPa");
    Serial.print(" | Elevasi (BMP): ");
    Serial.print(bmp.readAltitude(1013.25)); // Ketinggian berdasar tekanan laut standar
    Serial.print(" m");
    
    Serial.println();
  }

  // Peringatan jika lebih dari 5 detik modul GPS tidak merespon sama sekali
  static unsigned long lastGpsWarning = 0;
  if (millis() > 5000 && gps.charsProcessed() < 10) {
    if (millis() - lastGpsWarning > 5000) {
      Serial.println("Peringatan: Belum ada data dari GPS. Menunggu...");
      lastGpsWarning = millis();
    }
  }
}
